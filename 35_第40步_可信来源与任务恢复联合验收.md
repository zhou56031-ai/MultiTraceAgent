## 1. 验收要证明什么

这三项改造不能只凭页面好看就算完成。我们要分别证明：工具登记了实际来源、文件说明真实边界、报告包含来源、任务结果写入SQLite、通知丢失时前端仍能查询恢复，并且不重复执行或显示答案。

本页所有自动实验都明确替换了模型决策与搜索响应，不调用付费API。文件解析、正式工具、主任务函数、FastAPI、SQLite和浏览器交互仍实际运行。真实模型的行为需要另做少量真实验收，不能混写成这次模拟结果。

## 3. 新建后端联合验收脚本

### 新建 `learning/examples/40_reliability_check.py`

这是本步完整文件，写入以下内容：

```python
"""真实HTTP/SQLite/正式主任务函数，模拟图与搜索；不调用外部API。

直接运行做后端验收；--serve 启动临时验收服务供浏览器脚本使用。
"""
import asyncio
import os
import shutil
import sys
import time
import uuid
from contextlib import asynccontextmanager, ExitStack
from pathlib import Path
from tempfile import TemporaryDirectory
from unittest.mock import patch

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))
for key, value in {
    "OPENAI_API_KEY": "offline-test-placeholder", "TAVILY_API_KEY": "offline-test-placeholder",
    "OPENAI_BASE_URL": "https://example.invalid/v1", "LLM_QWEN_MAX": "offline-test-model",
}.items():
    os.environ[key] = value

from fastapi.testclient import TestClient
from langchain_core.messages import AIMessage
import agent.main_agent as runtime
import api.server as server
from api.context import get_session_context
from api.task_store import TaskStore
from tools.tavily_tool import client, internet_search
from tools.upload_file_read_tool import read_file_content
from tools.markdown_tools import generate_markdown


class FixtureGraph:
    """只替换不确定的模型决策，目录、工具、上下文、持久化都执行正式代码。"""
    def __init__(self):
        self.executions = 0

    async def astream(self, state, **kwargs):
        self.executions += 1
        query = state["messages"][0]["content"]
        if "wait" in query:
            await asyncio.sleep(30)
        if "slow" in query or "恢复验收" in query:
            await asyncio.sleep(1)
        if "fail" in query:
            raise ValueError("模拟失败")
        await asyncio.to_thread(internet_search.invoke, {"query": "模拟来源检索"})
        folder = Path(get_session_context())
        for file in sorted(folder.glob("*.txt")):
            await asyncio.to_thread(read_file_content.invoke, {"filename": file.name})
        await asyncio.to_thread(generate_markdown.invoke, {"filename": "验收报告", "content": "# 离线模拟报告\n结论使用模拟搜索资料 [S1]。"})
        yield {"model": {"messages": [AIMessage(content="离线模拟答案：任务结果可以找回，资料见 [S1]；文件读取范围见来源面板。") ]}}


@asynccontextmanager
async def fake_runtime():
    yield runtime.main_agent


def fixture(root: Path, graph: FixtureGraph):
    stack = ExitStack()
    for target, name, value in [
        (server, "task_store", TaskStore(root / "data/tasks.db")),
        (server, "upload_root", root / "updated"), (server, "output_root", root / "output"),
        (server, "agent_runtime", fake_runtime),
        (runtime, "project_root_path", root), (runtime, "main_agent", graph),
    ]:
        stack.enter_context(patch.object(target, name, value))
    stack.enter_context(patch.object(client, "search", return_value={"results": [
        {"title": "模拟来源", "url": "https://example.org/reliability", "content": "仅供离线验收的模拟摘要"},
    ]}))
    # 禁止真实出网。TestClient走进程内ASGI；--serve只需本机接受浏览器连接。
    stack.enter_context(patch("socket.socket.connect", side_effect=RuntimeError("离线验收禁止出网")))
    return stack


def wait_for(client: TestClient, thread: str, task: str, expected: str) -> dict:
    deadline = time.monotonic() + 8
    while time.monotonic() < deadline:
        response = client.get(f"/api/tasks/{task}", params={"thread_id": thread})
        response.raise_for_status()
        record = response.json()
        if record["status"] == expected:
            return record
        if record["status"] in {"failed", "cancelled", "interrupted"}:
            raise AssertionError(record)
        time.sleep(0.03)
    raise AssertionError("等待持久化状态超时")


def main():
    with TemporaryDirectory(prefix="learning-reliability-") as directory:
        root = Path(directory)
        graph = FixtureGraph()
        with fixture(root, graph):
            if "--serve" in sys.argv:
                import uvicorn
                print("模拟验收服务：http://127.0.0.1:8012；退出后清理全部临时数据", flush=True)
                uvicorn.run(server.app, host="127.0.0.1", port=8012, log_level="warning")
                return
            thread, task = str(uuid.uuid4()), str(uuid.uuid4())
            with TestClient(server.app) as client_api:
                uploaded = client_api.post("/api/upload", data={"thread_id": thread},
                                           files={"file": ("长文.txt", ("恢复验收\n" + "甲" * 13000).encode())})
                assert uploaded.status_code == 200
                body = {"query": "slow 恢复验收", "thread_id": thread, "task_id": task}
                submitted = client_api.post("/api/task", json=body)
                assert submitted.status_code == 202 and submitted.json()["created"]
                repeated = client_api.post("/api/task", json=body)
                assert not repeated.json()["created"]
                assert client_api.post("/api/task", json={**body, "task_id": str(uuid.uuid4())}).status_code == 409
                assert client_api.post("/api/task", json={**body, "query": "换一个问题"}).status_code == 409
                # 不连接WebSocket也能完成并找回结果。
                record = wait_for(client_api, thread, task, "succeeded")
                assert graph.executions == 1
                assert "本次工具来源" in record["result"] and "[S1]" in record["result"] and "[F1]" in record["result"]
                file_source = next(s for s in record["sources"] if s["kind"] == "file")
                assert file_source["char_truncated"] and file_source["returned_chars"] == 12000
                assert "仅供离线验收的模拟摘要" not in str(record["sources"])
                assert client_api.get(f"/api/tasks/{task}", params={"thread_id": str(uuid.uuid4())}).status_code == 404
                report = client_api.get("/api/download", params={"thread_id": thread, "filename": "验收报告.md"})
                assert report.status_code == 200 and "https://example.org/reliability" in report.text
                failed = client_api.post("/api/task", json={"query": "fail", "thread_id": thread}).json()["task_id"]
                assert wait_for(client_api, thread, failed, "failed")["error"] == "ValueError"
                cancelled = client_api.post("/api/task", json={"query": "wait", "thread_id": thread}).json()["task_id"]
            # 退出lifespan取消后台任务，结果落库。
            store = TaskStore(root / "data/tasks.db")
            assert store.get(cancelled, thread)["status"] == "cancelled"
            abandoned = str(uuid.uuid4())
            store.create(abandoned, thread, "模拟进程崩溃遗留任务")
            with TestClient(server.app) as reopened:
                assert reopened.get(f"/api/tasks/{task}", params={"thread_id": thread}).json()["result"] == record["result"]
                assert reopened.get(f"/api/tasks/{abandoned}", params={"thread_id": thread}).json()["status"] == "interrupted"
                assert reopened.get("/api/tasks", params={"thread_id": thread}).status_code == 200
    print("通过：HTTP受理与幂等、同会话并发保护、无WebSocket找回答案、真实工具来源/范围/报告、失败与取消、服务重开及崩溃遗留标记；图与搜索为模拟")


if __name__ == "__main__":
    main()
```

### 重要函数与变量，按调用顺序理解

| 名字 | 为什么存在、谁调用、输入输出 |
| --- | --- |
| FixtureGraph | 模拟模型节点，不模拟正式文件/来源/账本功能；executions统计实际启动几次图 |
| astream(state, **kwargs) | 正式run_deep_agent调用；根据测试query等待、失败或调用真实工具，最后yield一条模型消息 |
| fake_runtime | 替换服务启动时的外部能力探测，避免访问MySQL/RAGFlow与真实模型；yield期间使用FixtureGraph |
| fixture(root, graph) | 返回ExitStack，把服务数据路径、图对象与搜索响应替换到临时环境；退出时恢复原对象 |
| wait_for(client,thread,task,expected) | 循环GET精确任务，等到预期状态；终态不匹配或超时则断言失败 |
| main | 建临时目录→组装替换→运行真实API实验；--serve时改为临时HTTP服务 |

这里patch只是“把不可控外部服务换成固定练习材料”。socket.connect被阻止，避免脚本不小心发出真实请求。TestClient走进程内ASGI，不需要连8001；--serve另开8012，只接受浏览器请求，不改现有服务数据。

TestClient退出会执行真实lifespan清理，所以wait任务必须被取消并记录；再进入一次TestClient，会执行真实重启标记。因此不只是手动把数据库status改对再断言。

### 新建 `learning/examples/40_frontend_recovery.cjs`

这是本步完整文件，写入以下内容：

```javascript
/** 配合40_reliability_check.py --serve：真实HTTP/SQLite，屏蔽WS通知，模型与搜索为模拟。 */
const assert = require('node:assert/strict');
const fs = require('node:fs/promises');
const path = require('node:path');
const { chromium } = require(process.env.PLAYWRIGHT_MODULE || 'playwright');

(async () => {
  const browser = await chromium.launch({ headless: true,
    ...(process.env.BROWSER_PATH ? { executablePath: process.env.BROWSER_PATH } : {}) });
  try {
    const context = await browser.newContext({ viewport: { width: 1440, height: 1080 } });
    // 刻意不把任何完成事件传给网页；答案必须通过HTTP查回。
    const sockets = [];
    await context.routeWebSocket('**/ws/*', socket => { sockets.push(socket); });
    const page = await context.newPage();
    const errors = [], submitted = [];
    page.on('pageerror', error => errors.push(error.message));
    page.on('request', request => {
      if (request.method() === 'POST' && new URL(request.url()).pathname === '/api/task') submitted.push(request.postDataJSON());
    });
    const base = process.env.LEARNING_BASE_URL || 'http://127.0.0.1:8012';
    await page.goto(base);
    await page.getByText('实时连接', { exact: true }).waitFor();
    await page.locator('#uploadFile').setInputFiles({ name: '恢复验收.txt', mimeType: 'text/plain', buffer: Buffer.from('开始\n' + '甲'.repeat(13000)) });
    await page.locator('.upload-chip').filter({ hasText: '恢复验收.txt' }).waitFor();
    await page.locator('#query').fill('恢复验收：请分析文件，生成带来源的报告');
    const receipt = page.waitForResponse(response => new URL(response.url()).pathname === '/api/task' && response.request().method() === 'POST');
    await page.locator('#query').press('Enter');
    const response = await receipt;
    assert.equal(response.status(), 202);
    const accepted = await response.json();
    assert(accepted.task_id && accepted.thread_id && accepted.task_id !== accepted.thread_id);
    await page.reload();
    await page.locator('#status').filter({ hasText: '研究已完成' }).waitFor({ timeout: 15000 });
    assert.equal(submitted.length, 1, '刷新不能再次提交任务');
    assert.equal(await page.locator('.message.assistant').count(), 1);
    assert((await page.locator('.message.assistant').innerText()).includes('本次工具来源'));
    assert((await page.locator('#sources').innerText()).includes('正文已截断'));
    assert((await page.locator('#sources').innerText()).includes('[F1]'));
    assert.equal(await page.locator('#sources a[href="https://example.org/reliability"]').count(), 1);
    for (let index = 0; index < 3; index++) {
      const checked = page.waitForResponse(response => new URL(response.url()).pathname === '/api/tasks');
      await page.locator('#recoverTask').click(); await checked;
    }
    assert.equal(await page.locator('.message.assistant').count(), 1, '反复查回不能追加重复答案');
    sockets.at(-1).send(JSON.stringify({ type: 'monitor_event', task_id: '00000000-0000-4000-8000-000000000001', event: 'error', data: { detail: '晚到的其他任务错误' } }));
    assert.equal(await page.locator('#status').innerText(), '研究已完成');
    const download = await context.request.get(new URL('/api/download?' + new URLSearchParams({ thread_id: accepted.thread_id, filename: '验收报告.md' }), base).href);
    assert(download.ok() && (await download.text()).includes('https://example.org/reliability'));
    const directory = path.resolve(__dirname, '../output/ui-preview');
    await fs.mkdir(directory, { recursive: true });
    await page.locator('.toast').last().waitFor({state:'hidden',timeout:7000});
    await page.screenshot({ path: path.join(directory, 'recovery-desktop.png'), fullPage: true, animations: 'disabled' });
    await page.setViewportSize({ width: 390, height: 844 });
    assert(await page.evaluate(() => document.documentElement.scrollWidth <= innerWidth));
    await page.screenshot({ path: path.join(directory, 'recovery-mobile.png'), fullPage: true, animations: 'disabled' });
    assert.deepEqual(errors, []);
    console.log('通过：屏蔽WS通知后刷新由真实HTTP找回答案、只提交一次、重复核对不重复展示、来源与截断说明、报告下载、桌面/手机布局；模型和搜索为模拟');
  } finally { await browser.close(); }
})().catch(error => { console.error(error); process.exitCode = 1; });
```

这次故意把WebSocket通道截住，不给网页任何完成事件。提交、任务查询、上传与下载仍访问8012的真实HTTP接口。刷新后若页面能显示答案，证明答案是查回来的，不能靠“刚好收到了WS”蒙混通过。

submitted数组记录POST次数，刷新前后必须始终是1；`.message.assistant`必须始终是一条；反复点找回仍不新增答案。再发送另一个任务的迟到错误，当前成功状态不应被覆盖。最后检查桌面/390px手机布局、可打开来源、文件截断说明和报告下载。

## 7. 最终通过标准与实际记录

| 检查 | 应观察到的真实行为 |
| --- | --- |
| 文件边界 | 长文本明确截断、PDF前20页、Excel首表100行且只展示10行、Word不含表格 |

| 前端 | 刷新只查询、不重新POST；反复找回不重复答案；迟到消息不串任务 |

本轮实际运行结果在[全过程概要](00_复刻全过程概要.md)中更新。新增能力是来源目录与任务结果查询，不是自动逐句事实核查、完整文件解析、事件离线补发或任务自动续跑；后几项仍需独立设计与验收。
